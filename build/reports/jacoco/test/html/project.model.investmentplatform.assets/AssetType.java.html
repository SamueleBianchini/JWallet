package project.model.investmentplatform.assets;

/**
 * AssetType.
 * As we know, there are many type of assets in the market
 * where we can invest. The ones we choosed are:
 *      BOND: Fixed income securities (Treasury bonds, corporate bonds, etc.)
 *      STOCK: Equity securities (AAPL, GOOGL, MSFT, etc.)
 *      SPECULATIVE: High-risk investments (commodities, cryptocurrencies, etc.)
 */
public enum AssetType {
    BOUND,
    STOCK,
    SPECULATIVE
}
